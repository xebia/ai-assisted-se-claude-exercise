package library.reviews

import io.ktor.http.HttpStatusCode
import io.ktor.server.application.ApplicationCall
import io.ktor.server.plugins.NotFoundException
import io.ktor.server.response.respond
import io.ktor.server.routing.Route
import io.ktor.server.routing.delete
import io.ktor.server.routing.get
import io.ktor.server.routing.route
import kotlinx.serialization.Serializable
import library.events.EventBus
import org.jetbrains.exposed.dao.id.IntIdTable
import org.jetbrains.exposed.sql.ResultRow
import org.jetbrains.exposed.sql.and
import org.jetbrains.exposed.sql.javatime.CurrentTimestamp
import org.jetbrains.exposed.sql.javatime.timestamp
import org.jetbrains.exposed.sql.selectAll
import org.jetbrains.exposed.sql.transactions.transaction
import org.jetbrains.exposed.sql.update

object Reviews : IntIdTable("reviews") {
    val bookId = integer("book_id").index()
    val userId = varchar("user_id", 64)
    val rating = integer("rating")
    val body = varchar("body", 2000)
    val createdAt = timestamp("created_at").defaultExpression(CurrentTimestamp)
    val deletedAt = timestamp("deleted_at").nullable()
}

@Serializable
data class ReviewDto(val id: Int, val bookId: Int, val userId: String, val rating: Int, val body: String)

@Serializable
data class Problem(val type: String = "about:blank", val status: Int, val detail: String)

// Repository and routes live together: one module per resource.
class ReviewRepository {
    fun findActive(id: Int): ReviewDto? = transaction {
        Reviews.selectAll()
            .where { (Reviews.id eq id) and Reviews.deletedAt.isNull() }
            .map(::toDto)
            .singleOrNull()
    }

    // Never hard-delete: moderation needs the history.
    fun softDelete(id: Int) = transaction {
        Reviews.update({ (Reviews.id eq id) and Reviews.deletedAt.isNull() }) {
            it[deletedAt] = CurrentTimestamp
        }
    }

    private fun toDto(row: ResultRow) = ReviewDto(
        id = row[Reviews.id].value,
        bookId = row[Reviews.bookId],
        userId = row[Reviews.userId],
        rating = row[Reviews.rating],
        body = row[Reviews.body],
    )
}

fun Route.reviewRoutes(repo: ReviewRepository, events: EventBus) {
    route("/reviews/{reviewId}") {
        get {
            val id = call.reviewId() ?: return@get call.problem(HttpStatusCode.BadRequest, "invalid review id")
            val review = repo.findActive(id) ?: throw NotFoundException("review not found")
            call.respond(review)
        }

        // DELETE /reviews/{reviewId}
        // Only the author may delete. 204 on success, 404 if already gone.
        delete {
            val id = call.reviewId() ?: return@delete call.problem(HttpStatusCode.BadRequest, "invalid review id")
            val userId = call.request.headers["X-User-Id"]
                ?: return@delete call.problem(HttpStatusCode.Unauthorized, "missing X-User-Id")
            val review = repo.findActive(id)
                ?: return@delete call.problem(HttpStatusCode.NotFound, "review not found")
            if (review.userId != userId) {
                return@delete call.problem(HttpStatusCode.Forbidden, "not your review")
            }
            repo.softDelete(id)
            events.publish("review.deleted", mapOf("id" to id, "bookId" to review.bookId))
            call.respond(HttpStatusCode.NoContent)
        }
    }
}

private fun ApplicationCall.reviewId(): Int? = parameters["reviewId"]?.toIntOrNull()

private suspend fun ApplicationCall.problem(status: HttpStatusCode, detail: String) =
    respond(status, Problem(status = status.value, detail = detail))

// library-service/src/test/kotlin/library/reviews/ReviewApiTest.kt
package library.reviews

import io.kotest.core.spec.style.StringSpec
import io.kotest.matchers.shouldBe
import io.ktor.client.request.delete
import io.ktor.client.request.header
import io.ktor.http.HttpStatusCode
import io.ktor.server.routing.routing
import io.ktor.server.testing.testApplication
import io.mockk.every
import io.mockk.mockk
import io.mockk.verify

class ReviewApiTest : StringSpec({
    val repo = mockk<ReviewRepository>(relaxed = true)
    every { repo.findActive(7) } returns ReviewDto(7, 1, "u-42", 4, "good")

    "the author can delete and gets 204" {
        testApplication {
            application { routing { reviewRoutes(repo, mockk(relaxed = true)) } }
            client.delete("/reviews/7") { header("X-User-Id", "u-42") }
                .status shouldBe HttpStatusCode.NoContent
            verify { repo.softDelete(7) }
        }
    }

    "another user gets 403" {
        testApplication {
            application { routing { reviewRoutes(repo, mockk(relaxed = true)) } }
            client.delete("/reviews/7") { header("X-User-Id", "u-99") }
                .status shouldBe HttpStatusCode.Forbidden
        }
    }

    "no user header gets 401" {
        testApplication {
            application { routing { reviewRoutes(repo, mockk(relaxed = true)) } }
            client.delete("/reviews/7").status shouldBe HttpStatusCode.Unauthorized
        }
    }
})
