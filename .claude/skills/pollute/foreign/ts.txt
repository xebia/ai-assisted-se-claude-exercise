// library-service/src/reviews/api.ts
// Library service, reviews module (team Catalog). Shared for reference.
import { Hono } from "hono";
import { HTTPException } from "hono/http-exception";
import { zValidator } from "@hono/zod-validator";
import { z } from "zod";
import { and, eq, isNull } from "drizzle-orm";
import { integer, pgTable, serial, text, timestamp } from "drizzle-orm/pg-core";
import type { NodePgDatabase } from "drizzle-orm/node-postgres";
import { publish } from "../events";

export const reviews = pgTable("reviews", {
  id: serial("id").primaryKey(),
  bookId: integer("book_id").notNull(),
  userId: text("user_id").notNull(),
  rating: integer("rating").notNull(),
  body: text("body").notNull(),
  createdAt: timestamp("created_at").defaultNow(),
  deletedAt: timestamp("deleted_at"),
});

type Review = typeof reviews.$inferSelect;

// Repository and routes live together: one module per resource.
export class ReviewRepository {
  constructor(private db: NodePgDatabase) {}

  async findActive(id: number): Promise<Review | undefined> {
    const rows = await this.db
      .select()
      .from(reviews)
      .where(and(eq(reviews.id, id), isNull(reviews.deletedAt)));
    return rows[0];
  }

  // Never hard-delete: moderation needs the history.
  async softDelete(id: number): Promise<void> {
    await this.db
      .update(reviews)
      .set({ deletedAt: new Date() })
      .where(and(eq(reviews.id, id), isNull(reviews.deletedAt)));
  }
}

const idParam = z.object({ reviewId: z.coerce.number().int().positive() });

export function reviewRoutes(repo: ReviewRepository) {
  const app = new Hono();

  app.get("/reviews/:reviewId", zValidator("param", idParam), async (c) => {
    const { reviewId } = c.req.valid("param");
    const review = await repo.findActive(reviewId);
    if (!review) throw new HTTPException(404, { message: "review not found" });
    return c.json(review);
  });

  // DELETE /reviews/:reviewId
  // Only the author may delete. 204 on success, 404 if already gone.
  app.delete("/reviews/:reviewId", zValidator("param", idParam), async (c) => {
    const { reviewId } = c.req.valid("param");
    const userId = c.req.header("X-User-Id");
    if (!userId) throw new HTTPException(401, { message: "missing X-User-Id" });
    const review = await repo.findActive(reviewId);
    if (!review) throw new HTTPException(404, { message: "review not found" });
    if (review.userId !== userId) {
      throw new HTTPException(403, { message: "not your review" });
    }
    await repo.softDelete(reviewId);
    publish("review.deleted", { id: reviewId, bookId: review.bookId });
    return c.body(null, 204);
  });

  app.onError((err, c) => {
    const status = err instanceof HTTPException ? err.status : 500;
    return c.json({ type: "about:blank", status, detail: err.message }, status, {
      "Content-Type": "application/problem+json",
    });
  });

  return app;
}

// library-service/src/reviews/api.spec.ts
import { describe, it, expect, vi, beforeEach } from "vitest";
import { reviewRoutes, ReviewRepository } from "./api";

describe("DELETE /reviews/:reviewId", () => {
  let repo: ReviewRepository;

  beforeEach(() => {
    repo = {
      findActive: vi.fn().mockResolvedValue({ id: 7, bookId: 1, userId: "u-42" }),
      softDelete: vi.fn().mockResolvedValue(undefined),
    } as unknown as ReviewRepository;
  });

  it("lets the author delete and returns 204", async () => {
    const res = await reviewRoutes(repo).request("/reviews/7", {
      method: "DELETE",
      headers: { "X-User-Id": "u-42" },
    });
    expect(res.status).toBe(204);
    expect(repo.softDelete).toHaveBeenCalledWith(7);
  });

  it("returns 403 for another user", async () => {
    const res = await reviewRoutes(repo).request("/reviews/7", {
      method: "DELETE",
      headers: { "X-User-Id": "u-99" },
    });
    expect(res.status).toBe(403);
    expect(repo.softDelete).not.toHaveBeenCalled();
  });

  it("returns 401 without a user header", async () => {
    const res = await reviewRoutes(repo).request("/reviews/7", { method: "DELETE" });
    expect(res.status).toBe(401);
  });
});
