// library-service/internal/reviews/api.go
// Library service, reviews module (team Catalog). Shared for reference.
package reviews

import (
	"database/sql"
	"encoding/json"
	"errors"
	"log/slog"
	"net/http"
	"strconv"
	"time"

	"github.com/go-chi/chi/v5"
	"github.com/go-playground/validator/v10"
	"github.com/jmoiron/sqlx"
)

type Review struct {
	ID        int        `db:"id" json:"id"`
	BookID    int        `db:"book_id" json:"bookId"`
	UserID    string     `db:"user_id" json:"userId"`
	Rating    int        `db:"rating" json:"rating" validate:"min=1,max=5"`
	Body      string     `db:"body" json:"body" validate:"max=2000"`
	CreatedAt time.Time  `db:"created_at" json:"createdAt"`
	DeletedAt *time.Time `db:"deleted_at" json:"-"`
}

// Repository and handler live together: one module per resource.
type ReviewRepository struct{ db *sqlx.DB }

func (r *ReviewRepository) FindActive(id int) (*Review, error) {
	var rev Review
	err := r.db.Get(&rev,
		`SELECT * FROM reviews WHERE id = $1 AND deleted_at IS NULL`, id)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, nil
	}
	return &rev, err
}

// Never hard-delete: moderation needs the history.
func (r *ReviewRepository) SoftDelete(id int) error {
	_, err := r.db.Exec(
		`UPDATE reviews SET deleted_at = NOW() WHERE id = $1 AND deleted_at IS NULL`, id)
	return err
}

type Publisher interface{ Publish(topic string, payload any) }

type API struct {
	repo     *ReviewRepository
	events   Publisher
	validate *validator.Validate
}

func NewAPI(db *sqlx.DB, events Publisher) *API {
	return &API{repo: &ReviewRepository{db: db}, events: events, validate: validator.New()}
}

func (a *API) Routes(r chi.Router) {
	r.Get("/reviews/{reviewID}", a.GetReview)
	r.Delete("/reviews/{reviewID}", a.DeleteReview)
}

func (a *API) GetReview(w http.ResponseWriter, r *http.Request) {
	id, err := strconv.Atoi(chi.URLParam(r, "reviewID"))
	if err != nil {
		writeProblem(w, http.StatusBadRequest, "invalid review id")
		return
	}
	rev, err := a.repo.FindActive(id)
	if err != nil {
		writeProblem(w, http.StatusInternalServerError, "lookup failed")
		return
	}
	if rev == nil {
		writeProblem(w, http.StatusNotFound, "review not found")
		return
	}
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(rev)
}

// DELETE /reviews/{reviewID}
// Only the author may delete. 204 on success, 404 if already gone.
func (a *API) DeleteReview(w http.ResponseWriter, r *http.Request) {
	id, err := strconv.Atoi(chi.URLParam(r, "reviewID"))
	if err != nil {
		writeProblem(w, http.StatusBadRequest, "invalid review id")
		return
	}
	userID := r.Header.Get("X-User-Id")
	if userID == "" {
		writeProblem(w, http.StatusUnauthorized, "missing X-User-Id")
		return
	}
	rev, err := a.repo.FindActive(id)
	if err != nil {
		writeProblem(w, http.StatusInternalServerError, "lookup failed")
		return
	}
	if rev == nil {
		writeProblem(w, http.StatusNotFound, "review not found")
		return
	}
	if rev.UserID != userID {
		writeProblem(w, http.StatusForbidden, "not your review")
		return
	}
	if err := a.repo.SoftDelete(id); err != nil {
		writeProblem(w, http.StatusInternalServerError, "delete failed")
		return
	}
	a.events.Publish("review.deleted", map[string]any{"id": id, "bookId": rev.BookID})
	slog.Info("review deleted", "id", id, "user", userID)
	w.WriteHeader(http.StatusNoContent)
}

func writeProblem(w http.ResponseWriter, status int, detail string) {
	w.Header().Set("Content-Type", "application/problem+json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(map[string]any{
		"type": "about:blank", "status": status, "detail": detail,
	})
}

// library-service/internal/reviews/api_test.go
package reviews

import (
	"net/http"
	"net/http/httptest"
	"testing"

	"github.com/DATA-DOG/go-sqlmock"
	"github.com/go-chi/chi/v5"
	"github.com/jmoiron/sqlx"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

type noopPublisher struct{}

func (noopPublisher) Publish(string, any) {}

func TestDeleteReview_OwnerGets204(t *testing.T) {
	db, mock, err := sqlmock.New()
	require.NoError(t, err)
	rows := sqlmock.NewRows([]string{"id", "book_id", "user_id", "rating", "body"}).
		AddRow(7, 1, "u-42", 4, "good")
	mock.ExpectQuery("SELECT \\* FROM reviews").WillReturnRows(rows)
	mock.ExpectExec("UPDATE reviews SET deleted_at").WillReturnResult(sqlmock.NewResult(0, 1))

	r := chi.NewRouter()
	NewAPI(sqlx.NewDb(db, "postgres"), noopPublisher{}).Routes(r)
	req := httptest.NewRequest(http.MethodDelete, "/reviews/7", nil)
	req.Header.Set("X-User-Id", "u-42")
	rec := httptest.NewRecorder()
	r.ServeHTTP(rec, req)

	assert.Equal(t, http.StatusNoContent, rec.Code)
	assert.NoError(t, mock.ExpectationsWereMet())
}

func TestDeleteReview_OtherUserGets403(t *testing.T) {
	db, mock, _ := sqlmock.New()
	rows := sqlmock.NewRows([]string{"id", "book_id", "user_id", "rating", "body"}).
		AddRow(7, 1, "u-42", 4, "good")
	mock.ExpectQuery("SELECT \\* FROM reviews").WillReturnRows(rows)

	r := chi.NewRouter()
	NewAPI(sqlx.NewDb(db, "postgres"), noopPublisher{}).Routes(r)
	req := httptest.NewRequest(http.MethodDelete, "/reviews/7", nil)
	req.Header.Set("X-User-Id", "u-99")
	rec := httptest.NewRecorder()
	r.ServeHTTP(rec, req)

	assert.Equal(t, http.StatusForbidden, rec.Code)
}
